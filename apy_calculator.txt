#include <iostream> 
#include <string> 
#include <sstream> // to read commas and the percentage signs
#include <cmath> // to use exponents 
#include <algorithm> // to format with commas

using namespace std; 

double parseAmount(string input) { // this is to read commas 
    string noCommas = "";
    for (char c : input) {
    if (c != ',') noCommas += c; }
    stringstream ss(noCommas); 
    double value;
    ss >> value;
    return value; }

double parsePercent(string input) { // this is to read percentage signs
    if (!input.empty() && input.back() == '%') input.pop_back();
    return stod(input); }
    
std::string formatWithCommas(int number) { // to format with commas
    std::string numStr = std::to_string(number);
    int insertPosition = numStr.length() - 3;
    while (insertPosition > 0) {
        numStr.insert(insertPosition, ",");
        insertPosition -= 3; }
    return numStr; }
